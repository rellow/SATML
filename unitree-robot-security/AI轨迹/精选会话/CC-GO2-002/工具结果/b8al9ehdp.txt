# 打印错误信息（红色）
print_error() {
    if [ "$QUIET" != true ]; then
        echo -e "${RED}[ERROR]${NC} $1"
    fi
    log_message "ERROR: $1"
    log_to_script "ERROR: $1"
}

print_success() {
    if [ "$QUIET" != true ]; then
        echo -e "${GREEN}[SUCCESS]${NC} $1"
    fi
    log_message "SUCCESS: $1"
    log_to_script "SUCCESS: $1"
}

print_info() {
    if [ "$QUIET" != true ]; then
        echo -e "${YELLOW}[INFO]${NC} $1"
    fi
    log_message "INFO: $1"
    log_to_script "INFO: $1"
}

print_debug() {
    if [ "$QUIET" != true ]; then
        echo -e "${BLUE}[DEBUG]${NC} $1"
    fi
    log_message "DEBUG: $1"
    log_to_script "DEBUG: $1"
}

# Log to WPA persistent file (only when KEEP_LOGS=true)
log_message() {
    if [ "$KEEP_LOGS" = true ] && [ -n "$PERSISTENT_LOG" ]; then
        echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$PERSISTENT_LOG" 2>/dev/null || true
    fi
}

# Log script messages to script log file (always active)
log_to_script() {
    if [ -n "$SCRIPT_LOG" ]; then
        echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" >> "$SCRIPT_LOG" 2>/dev/null
    fi
}

#############################################
# 文件路径管理模块
# ============================================
# 功能：基于SSID生成安全的文件路径
# - 将SSID转换为安全的文件名（去除特殊字符）
# - 使用MD5哈希确保唯一性
# - 生成配置文件、日志文件等路径
#############################################

# 基于SSID创建文件路径
# 参数：$1 - SSID名称
# 功能：将SSID转换为安全的文件名，并设置全局变量：
#       WPA_CONF, WPA_LOG, SCRIPT_LOG, PERSISTENT_LOG, PERSISTENT_SCRIPT_LOG
create_ssid_based_paths() {
    local ssid="$1"
    local safe_ssid
    local ssid_hash
    
    if [ -z "$ssid" ]; then
        print_error "SSID cannot be empty for path creation"
        return 1
    fi
    
    # Sanitize SSID for filename (replace spaces and special chars with underscores)
    # Limit length to prevent filesystem issues
    safe_ssid=$(echo "$ssid" | sed 's/[^a-zA-Z0-9._-]/_/g' | cut -c1-32)
    
    # Generate a short hash of the original SSID for uniqueness
    # This prevents collisions when different SSIDs produce the same safe_ssid
    ssid_hash=$(echo -n "$ssid" | md5sum 2>/dev/null | cut -c1-8)
    
    # If hash generation fails (no md5sum), use a simpler approach
    if [ -z "$ssid_hash" ]; then
        ssid_hash=$(echo -n "$ssid" | cksum | cut -d' ' -f1)
    fi
    
    # Combine safe_ssid with hash for uniqueness
    # If safe_ssid is empty or only underscores, use "ssid" prefix
    if [ -z "$safe_ssid" ] || [ "$safe_ssid" = "_" ] || [[ "$safe_ssid" =~ ^_+$ ]]; then
        safe_ssid="ssid_${ssid_hash}"
    else
        safe_ssid="${safe_ssid}_${ssid_hash}"
    fi
    
    # Set file paths based on SSID
    WPA_CONF="${SCRIPT_DIR}/wpa_${safe_ssid}.conf"
    WPA_LOG="${SCRIPT_DIR}/wpa_${safe_ssid}.log"
    SCRIPT_LOG="${SCRIPT_DIR}/wpa_script_${safe_ssid}.log"
    PERSISTENT_LOG="/var/log/wpa_${safe_ssid}.log"
    PERSISTENT_SCRIPT_LOG="/var/log/wpa_script_${safe_ssid}.log"
    
    if [ "$DEBUG" = true ]; then
        print_debug "Using SSID-based file paths:"
        print_debug "Original SSID: $ssid"
        print_debug "Safe SSID: $safe_ssid"
        print_debug "Config: $WPA_CONF"
        print_debug "WPA Log: $WPA_LOG"
        print_debug "Script Log: $SCRIPT_LOG"
        print_debug "Persistent WPA: $PERSISTENT_LOG"
        print_debug "Persistent Script: $PERSISTENT_SCRIPT_LOG"
    fi
}

# Clean previous connection files for a given SSID
clean_previous_ssid_files() {
    local ssid="$1"
    local remove_persistent="${2:-false}"
    local safe_ssid
    local ssid_hash
    
    if [ -z "$ssid" ]; then
        return 0
    fi
    
    # Sanitize SSID for filename (consistent with create_ssid_based_paths)
    safe_ssid=$(echo "$ssid" | sed 's/[^a-zA-Z0-9._-]/_/g' | cut -c1-32)
    
    # Generate the same hash as in create_ssid_based_paths
    ssid_hash=$(echo -n "$ssid" | md5sum 2>/dev/null | cut -c1-8)
    
    # If hash generation fails (no md5sum), use cksum
    if [ -z "$ssid_hash" ]; then
        ssid_hash=$(echo -n "$ssid" | cksum | cut -d' ' -f1)
    fi
    
    # Combine safe_ssid with hash (same logic as create_ssid_based_paths)
    if [ -z "$safe_ssid" ] || [ "$safe_ssid" = "_" ] || [[ "$safe_ssid" =~ ^_+$ ]]; then
        safe_ssid="ssid_${ssid_hash}"
    else
        safe_ssid="${safe_ssid}_${ssid_hash}"
    fi
    
    # Clean previous temp files for this SSID
    rm -f "${SCRIPT_DIR}/wpa_${safe_ssid}.conf" "${SCRIPT_DIR}/wpa_${safe_ssid}.log" "${SCRIPT_DIR}/wpa_script_${safe_ssid}.log" 2>/dev/null
    
    # Clean any error files that might exist
    rm -f "${SCRIPT_DIR}/wpa_error_${safe_ssid}"*.txt 2>/dev/null
    
    # Optionally clean persistent logs created for this SSID
    if [ "$remove_persistent" = true ]; then
        rm -f "/var/log/wpa_${safe_ssid}.log" "/var/log/wpa_script_${safe_ssid}.log" 2>/dev/null
    fi
    
    if [ "$DEBUG" = true ]; then
        if [ "$remove_persistent" = true ]; then
            print_debug "Cleaned previous (temp + persistent) files for SSID: $ssid"
        else
            print_debug "Cleaned previous temporary files for SSID: $ssid"
        fi
    fi
}

# Clean logs generated during the previous script execution (regardless of new SSID)
clean_last_session_logs() {
    if [ ! -f "$LAST_SESSION_FILE" ]; then
        return 0
    fi
    
    local last_ssid
    last_ssid=$(cat "$LAST_SESSION_FILE" 2>/dev/null)
    
    if [ -n "$last_ssid" ]; then
        print_info "Cleaning previous session logs (SSID: $last_ssid)"
        clean_previous_ssid_files "$last_ssid" false
    fi
    
    rm -f "$LAST_SESSION_FILE" 2>/dev/null
}

# Record the SSID for the current session so it can be cleaned on the next run
record_current_session() {
    local ssid="$1"
    echo "$ssid" > "$LAST_SESSION_FILE" 2>/dev/null
}

#############################################
# 单实例锁机制模块 (SINGLE INSTANCE LOCK MECHANISM)
# ============================================
# 功能：确保同一时间只有一个脚本实例在运行，防止并发冲突
#
# 锁文件位置：${SCRIPT_DIR}/.wpa_connect.lock
# 锁文件内容：当前进程的PID
# 锁超时时间：LOCK_TIMEOUT=300秒（防止死锁）
#
# 工作流程：
# 1. acquire_lock() - 尝试获取锁
#    - 检查锁文件是否存在
#    - 如果存在，检查是否为过期锁（进程已不存在）
#    - 如果锁有效，等待或根据操作类型处理
#    - 尝试创建锁文件（使用原子操作 set -C）
#    - 最多重试10次，每次间隔0.3-0.5秒
#
# 2. check_lock_stale() - 检查锁是否过期
#    - 检查锁文件是否存在
#    - 读取锁文件中的PID
#    - 检查进程是否还在运行
#    - 检查进程是否为wpa_connect.sh脚本
#    - 返回：0=过期，1=有效
#
# 3. release_lock() - 释放锁
#    - 检查锁文件是否存在
#    - 验证锁文件中的PID是否为当前进程
#    - 只有当前进程的锁才会被释放（防止误删其他进程的锁）
#    - 删除锁文件
#
# 特殊处理：
# - disconnect操作：如果检测到其他实例，会终止它以便执行断开操作
# - connect操作：如果检测到其他实例，会等待其完成或超时退出
#
#############################################

# 检查锁文件是否过期（进程是否还存在）
# 返回值：0=锁已过期（可删除），1=锁有效（进程还在运行）
check_lock_stale() {
    if [ ! -f "$LOCK_FILE" ]; then
        return 1  # No lock file
    fi
    
    local lock_pid
    lock_pid=$(cat "$LOCK_FILE" 2>/dev/null | head -n1)
    
    if [ -z "$lock_pid" ]; then
        return 0  # Lock file is empty, consider it stale
    fi
    
    # Check if process exists
    if ! ps -p "$lock_pid" > /dev/null 2>&1; then
        return 0  # Process doesn't exist, lock is stale
    fi
    
    # Check if it's actually our script
    if ! ps -p "$lock_pid" -o cmd= 2>/dev/null | grep -q "wpa_connect.sh"; then
        return 0  # Not our script, lock is stale
    fi
    
    return 1  # Lock is valid
}

# 获取锁（带超时和重试机制）
# 功能：尝试获取单实例锁，防止多个脚本实例同时运行
# 策略：
#   - 最多尝试10次获取锁
#   - 每次尝试间隔0.3-0.5秒
#   - 如果锁文件存在但已过期（进程不存在），自动清理
#   - 如果锁有效且是connect操作，等待其他实例完成
#   - 如果锁有效且是disconnect操作，终止其他实例
# 返回值：0=成功获取锁，1=获取锁失败
acquire_lock() {
    local max_attempts=50  # 最大尝试次数
    local attempt=0        # 当前尝试次数
    
    # 循环尝试获取锁
    while [ $attempt -lt $max_attempts ]; do
        # Check if lock file exists and is valid
        if [ -f "$LOCK_FILE" ]; then
            if check_lock_stale; then
                # Lock is stale, remove it
                if [ "$DEBUG" = true ]; then
                    print_debug "Removing stale lock file"
                fi
                rm -f "$LOCK_FILE" 2>/dev/null
            else
                # Lock is valid, check if we should wait or exit
                local lock_pid
                lock_pid=$(cat "$LOCK_FILE" 2>/dev/null | head -n1)
                
                if [ -n "$lock_pid" ]; then
                    if [ "$DEBUG" = true ]; then
                        print_debug "Another instance is running (PID: $lock_pid)"
                    fi
                    
                    # For disconnect action, allow it to proceed (might need to clean up)
                    if [ "$ACTION" = "disconnect" ]; then
                        print_info "Another instance detected (PID: $lock_pid)"
                        print_info "Terminating previous instance to allow disconnect..."
                        # Try to kill the other instance
                        if ps -p "$lock_pid" > /dev/null 2>&1; then
                            kill -TERM "$lock_pid" 2>/dev/null
                            sleep 1
                            if ps -p "$lock_pid" > /dev/null 2>&1; then
                                kill -KILL "$lock_pid" 2>/dev/null
                                sleep 0.5
                            fi
                        fi
                        # Remove the lock file and try again
                        rm -f "$LOCK_FILE" 2>/dev/null
                        sleep 0.3
                        continue
                    else
                        # For connect action, wait a bit and retry
                        if [ $attempt -eq 0 ]; then
                            print_info "Another instance is running (PID: $lock_pid), waiting..."
                        fi
                        
                        if [ $attempt -lt $((max_attempts - 1)) ]; then
                            if [ "$QUIET" != true ]; then
                                echo -n "."
                            fi
                            sleep 0.5
                            attempt=$((attempt + 1))
                            continue
                        else
                            print_error "Another instance is still running (PID: $lock_pid)"
                            print_error "Previous instance may be stuck or taking too long"
                            print_info "Options:"
                            print_info "  1. Wait for the previous instance to complete"
                            print_info "  2. Kill the stuck instance: sudo kill $lock_pid"
                            print_info "  3. Force remove lock: sudo rm -f $LOCK_FILE"
                            print_info "  4. Check what it's doing: ps aux | grep $lock_pid"
                            exit $EXIT_GENERAL_ERROR
                        fi
                    fi
                fi
            fi
        fi
        
        # Try to create lock file
        if (set -C; echo $$ > "$LOCK_FILE") 2>/dev/null; then
            # Lock acquired successfully
            if [ "$DEBUG" = true ]; then
                print_debug "Lock acquired (PID: $$)"
            fi
            return 0
        fi
        
        # Lock creation failed, wait and retry
        sleep 0.3
        attempt=$((attempt + 1))
    done
    
    # Failed to acquire lock
    print_error "Failed to acquire lock after $max_attempts attempts"
    return 1
}

# 释放锁
# 功能：安全地释放当前进程持有的锁
# 安全机制：
#   - 只释放属于当前进程（$$）的锁
#   - 防止误删其他进程的锁文件
#   - 如果锁文件不存在或属于其他进程，静默返回
# 使用场景：
#   - 脚本正常退出时（通过trap EXIT调用）
#   - 脚本被中断时（通过trap INT TERM调用）
release_lock() {
    if [ -f "$LOCK_FILE" ]; then
        local lock_pid
        lock_pid=$(cat "$LOCK_FILE" 2>/dev/null | head -n1)
        
        # 只删除属于当前进程的锁（安全机制）
        if [ "$lock_pid" = "$$" ]; then
            rm -f "$LOCK_FILE" 2>/dev/null
            if [ "$DEBUG" = true ]; then
                print_debug "Lock released (PID: $$)"
            fi
        else
            if [ "$DEBUG" = true ]; then
                print_debug "Lock file belongs to another process (PID: $lock_pid), not removing"
            fi
        fi
    fi
}

# 配置文件管理模块
# ============================================
# 功能：管理WPA配置文件的查找和提取
# - 查找上次生成的配置文件（用于重连）
# - 从配置文件提取SSID信息
#############################################

# 查找并返回上次生成的配置文件
# 参数：$1 - 网络接口名称
# 返回值：配置文件路径（成功）或空（失败）
# 查找策略：
#   1. 首先从LAST_SESSION_FILE读取上次的SSID
#   2. 基于SSID生成预期的配置文件路径
#   3. 如果失败，查找最新的wpa_*.conf文件
find_last_conf_file() {
    local interface="$1"
    
    # First try to get SSID from last session file
    if [ -f "$LAST_SESSION_FILE" ]; then
        local last_ssid
        last_ssid=$(cat "$LAST_SESSION_FILE" 2>/dev/null)
        
        if [ -n "$last_ssid" ]; then
            # Generate the expected conf file path for this SSID
            local safe_ssid
            local ssid_hash
            
            safe_ssid=$(echo "$last_ssid" | sed 's/[^a-zA-Z0-9._-]/_/g' | cut -c1-32)
            ssid_hash=$(echo -n "$last_ssid" | md5sum 2>/dev/null | cut -c1-8)
            
            if [ -z "$ssid_hash" ]; then
                ssid_hash=$(echo -n "$last_ssid" | cksum | cut -d' ' -f1)
            fi
            
            if [ -z "$safe_ssid" ] || [ "$safe_ssid" = "_" ] || [[ "$safe_ssid" =~ ^_+$ ]]; then
                safe_ssid="ssid_${ssid_hash}"
            else
                safe_ssid="${safe_ssid}_${ssid_hash}"
            fi
            
            local conf_file="${SCRIPT_DIR}/wpa_${safe_ssid}.conf"
            
            if [ -f "$conf_file" ]; then
                echo "$conf_file"
                return 0
            fi
        fi
    fi
    
    # Fallback: find the most recent conf file
    local latest_conf
    latest_conf=$(ls -t "${SCRIPT_DIR}"/wpa_*.conf 2>/dev/null | head -n1)
    
    if [ -n "$latest_conf" ] && [ -f "$latest_conf" ]; then
        echo "$latest_conf"
        return 0
    fi
    
    return 1
}

# Extract SSID from conf file
extract_ssid_from_conf() {
    local conf_file="$1"
    
    if [ ! -f "$conf_file" ]; then
        return 1
    fi
    
    # Extract SSID from network block
    local ssid
    ssid=$(grep -E '^\s*ssid=' "$conf_file" | head -n1 | sed -E 's/^\s*ssid="?([^"]*)"?\s*$/\1/')
    
    if [ -n "$ssid" ]; then
        echo "$ssid"
        return 0
    fi
    
    return 1
}

#############################################
# 进程管理模块
# ============================================
# 功能：管理进程的停止和等待
# - 优雅地停止进程（先TERM，后KILL）
# - 等待进程完全停止
# - 支持按模式匹配进程
#############################################

# 等待进程停止
# 参数：
#   $1 - 进程匹配模式（用于pgrep -f）
#   $2 - 最大等待时间（秒，默认PROCESS_STOP_TIMEOUT）
# 返回值：0=进程已停止，1=超时进程仍在运行
wait_for_process_stop() {
    local process_pattern="$1"
    local max_wait="${2:-$PROCESS_STOP_TIMEOUT}"
    local count=0
    
    if [ -z "$process_pattern" ]; then
        return 1
    fi
    
    while pgrep -f "$process_pattern" > /dev/null 2>&1 && [ $count -lt $((max_wait * 10)) ]; do
        sleep 0.1
        count=$((count + 1))
    done
    
    # Check if process still exists
    if pgrep -f "$process_pattern" > /dev/null 2>&1; then
        return 1  # Process still running
    fi
    
    return 0  # Process stopped successfully
}

# 按模式停止进程（优雅终止 + 强制终止）
# 参数：
#   $1 - 进程匹配模式（用于pgrep -f）
#   $2 - 初始信号（默认TERM）
#   $3 - 最大等待时间（秒，默认PROCESS_STOP_TIMEOUT）
#   $4 - 是否强制终止（默认true，TERM失败后使用KILL）
# 返回值：0=成功停止，1=停止失败
# 策略：
#   1. 发送TERM信号（优雅终止）
#   2. 等待进程停止
#   3. 如果仍运行且force_kill=true，发送KILL信号
stop_process_by_pattern() {
    local pattern="$1"
    local signal="${2:-TERM}"
    local max_wait="${3:-$PROCESS_STOP_TIMEOUT}"
    local force_kill="${4:-true}"  # Whether to use KILL if TERM fails
    
    if ! pgrep -f "$pattern" > /dev/null 2>&1; then
        return 0  # Process not running
    fi
    
    if [ "$DEBUG" = true ]; then
        print_debug "Stopping processes matching: $pattern"
    fi
    
    # Get PIDs before killing
    local pids
    pids=$(pgrep -f "$pattern" 2>/dev/null)
    
    # Send initial signal (TERM by default)
    pkill -$signal -f "$pattern" 2>/dev/null
    
    # Wait for processes to stop
    if wait_for_process_stop "$pattern" "$max_wait"; then
        if [ "$DEBUG" = true ]; then
            print_debug "Processes stopped gracefully"
        fi
        return 0
    fi
    
    # If force_kill is enabled and processes are still running, use KILL
    if [ "$force_kill" = true ] && pgrep -f "$pattern" > /dev/null 2>&1; then
        if [ "$DEBUG" = true ]; then
            print_debug "Processes did not stop, forcing kill..."
        fi
        pkill -KILL -f "$pattern" 2>/dev/null
        sleep 0.5  # Give it a moment
        
        # Final check
        if pgrep -f "$pattern" > /dev/null 2>&1; then
            print_error "Failed to stop processes matching: $pattern"
            return 1
        else
            if [ "$DEBUG" = true ]; then
                print_debug "Processes forcefully terminated"
            fi
            return 0
        fi
    fi
    
    return 1
}

#############################################
# 网络接口管理模块
# ============================================
# 功能：检查和管理网络接口状态
# - 验证接口是否存在
# - 检查接口是否为无线接口
# - 自动启动DOWN状态的接口
#############################################

# 检查接口状态
# 参数：$1 - 网络接口名称
# 返回值：
#   0 = 接口正常（存在且为无线接口）
#   1 = 接口不存在
#   2 = 接口启动失败
#   3 = 接口不是无线接口
check_interface_status() {
    local interface="$1"
    
    if [ -z "$interface" ]; then
        print_error "Interface parameter cannot be empty"
        return 1
    fi
    
    # Check if interface exists
    if ! ip link show "$interface" &> /dev/null; then
        print_error "Network interface '$interface' does not exist"
        if [ "$QUIET" != true ]; then
            print_info "Available interfaces:"
            ip link show | grep -E "^[0-9]+:" | awk '{print $2}' | sed 's/://' | grep -v '^lo$'
        fi
        return 1
    fi
    
    # Check if interface is up
    local interface_state
    interface_state=$(ip link show "$interface" | grep -o 'state [A-Z]*' | awk '{print $2}')
    if [ "$interface_state" = "DOWN" ]; then
        print_info "Interface '$interface' is down, attempting to bring it up..."
        if ! ip link set "$interface" up 2>/dev/null; then
            print_error "Failed to bring interface '$interface' up"
            return 2
        fi
        sleep 0.3  # 从1秒减少到0.3秒
    fi
    
    # Check if interface is wireless
    if ! iw dev "$interface" info &> /dev/null; then
        print_error "Interface '$interface' is not a wireless interface"
        return 3
    fi
    
    return 0
}

# 手动生成WPA配置（当wpa_passphrase失败时的备用方法）

#   $3 - 配置文件路径

# 生成WPA配置（多种方法，自动降级）
# 参数：
#   $1 - SSID
#   $2 - 密码
#   $3 - 配置文件路径
# 返回值：0=成功，1=失败
# 生成策略（按优先级）：
#   1. 使用wpa_passphrase直接参数方式
#   2. 使用wpa_passphrase标准输入方式
#   3. 手动生成配置（备用方法）
generate_wpa_config() {
    local ssid="$1"
    local password="<密码_01>"
    local config_file="$3"
    local safe_ssid
    local ssid_hash
    local error_file
    
    # Create SSID-based error file with hash for uniqueness
    safe_ssid=$(echo "$ssid" | sed 's/[^a-zA-Z0-9._-]/_/g' | cut -c1-32)
    ssid_hash=$(echo -n "$ssid" | md5sum 2>/dev/null | cut -c1-8)
    
    if [ -z "$ssid_hash" ]; then
        ssid_hash=$(echo -n "$ssid" | cksum | cut -d' ' -f1)
    fi
    
    if [ -z "$safe_ssid" ] || [[ "$safe_ssid" =~ ^_+$ ]]; then
        safe_ssid="ssid_${ssid_hash}"
    else
        safe_ssid="${safe_ssid}_${ssid_hash}"
    fi
    
    error_file="${SCRIPT_DIR}/wpa_error_${safe_ssid}.txt"
    TEMP_FILES_TO_CLEAN="$TEMP_FILES_TO_CLEAN $error_file"
    
    # Method 1: Direct arguments
    if wpa_passphrase "$ssid" "$password" > "$config_file" 2>"$error_file"; then
        if [ -s "$config_file" ]; then
            # Add scan_ssid=1 to the network block for faster hidden SSID detection
            sed -i '/^[[:space:]]*psk=/a\    scan_ssid=1' "$config_file"
            
            # Add additional config
            cat >> "$config_file" << 'EOF'

ctrl_interface=/var/run/wpa_supplicant
update_config=1
ap_scan=1
fast_reauth=1
bgscan="simple:30:-70:300"
EOF
            rm -f "$error_file"
            print_success "Configuration generated successfully"
            return 0
        fi
    fi
    
    # Method 2: Stdin
    if [ "$DEBUG" = true ]; then
        print_debug "Method 1 failed, trying stdin method..."
    fi
    
    if printf "%s\n%s\n" "$ssid" "$password" | wpa_passphrase > "$config_file" 2>"$error_file"; then
        if [ -s "$config_file" ]; then
            # Add scan_ssid=1 to the network block for faster hidden SSID detection
            sed -i '/^[[:space:]]*psk=/a\    scan_ssid=1' "$config_file"
            
            # Add additional config
            cat >> "$config_file" << 'EOF'

ctrl_interface=/var/run/wpa_supplicant
update_config=1
ap_scan=1
fast_reauth=1
bgscan="simple:30:-70:300"
EOF
            rm -f "$error_file"
            print_success "Configuration generated successfully"
            return 0
        fi
    fi
    
    # Method 3: Manual generation
    if [ "$DEBUG" = true ]; then
        print_debug "wpa_passphrase failed, using manual generation..."
        if [ -s "$error_file" ]; then
            print_debug "Error details:"
            cat "$error_file" | sed 's/^/  /'
        fi
    fi
    
    generate_manual_config "$ssid" "$password" "$config_file"
    
    if [ -s "$config_file" ]; then
        print_success "Manual configuration generated successfully"
        rm -f "$error_file"
        return 0
    fi
    
    rm -f "$error_file"
    return 1
}

#############################################
# DHCP地址获取模块
# ============================================
# 功能：获取DHCP分配的IP地址
# - 支持多种DHCP客户端（dhclient、dhcpcd、udhcpc）
# - 自动检测可用的DHCP客户端
# - 支持重试机制（最多3次）
# - 快速验证IP分配和网关
#############################################

# 获取DHCP地址（支持多种客户端和重试机制）
# 参数：$1 - 网络接口名称
# 返回值：0=成功获取IP，1=失败
# 支持的DHCP客户端（按优先级）：
#   1. dhclient
#   2. dhcpcd
#   3. udhcpc
get_dhcp_address() {
    local interface="$1"
    local dhcp_clients=("dhclient -v" "dhcpcd" "udhcpc -i")
    local dhcp_commands=("dhclient" "dhcpcd" "udhcpc")
    local timeout_duration=10  # 从15秒减少到10秒
    local retry_count=0
    
    if [ -z "$interface" ]; then
        print_error "Interface parameter cannot be empty for DHCP"
        return 1
    fi
    
    # Flush existing IP first
    ip addr flush dev "$interface" 2>/dev/null
    
    while [ $retry_count -lt $MAX_DHCP_RETRIES ]; do
        for i in "${!dhcp_commands[@]}"; do
            if command -v "${dhcp_commands[$i]}" &> /dev/null; then
                print_info "Using ${dhcp_commands[$i]} for DHCP (attempt $((retry_count + 1))/$MAX_DHCP_RETRIES)..."
                
                # Kill any existing DHCP clients for this interface first
                pkill -f "${dhcp_commands[$i]}.*$interface" 2>/dev/null || true
                sleep 0.2  # 从0.5秒减少到0.2秒
                
                if timeout "$timeout_duration" ${dhcp_clients[$i]} "$interface" &> /dev/null; then
                    # 快速检查IP是否已分配
                    local max_check=10
                    local check_count=0
                    while [ $check_count -lt $max_check ]; do
                        local ip_addr
                        ip_addr=$(ip addr show "$interface" | grep "inet " | awk '{print $2}' | head -n1)
                        
                        if [ -n "$ip_addr" ]; then
                            print_success "IP address obtained: $ip_addr"
                            # Verify connectivity by checking routing table
                            local gateway
                            gateway=$(ip route | grep default | grep "$interface" | awk '{print $3}' | head -n1)
                            if [ -n "$gateway" ]; then
                                print_info "Default gateway: $gateway"
                            fi
                            return 0
                        fi
                        sleep 0.2
                        check_count=$((check_count + 1))
                    done
                fi
                
                if [ "$DEBUG" = true ]; then
                    print_debug "${dhcp_commands[$i]} failed or timed out"
                fi
                # Clean up failed DHCP client
                pkill -f "${dhcp_commands[$i]}.*$interface" 2>/dev/null || true
            fi
        done
        retry_count=$((retry_count + 1))
        
        if [ $retry_count -lt $MAX_DHCP_RETRIES ]; then
            print_info "DHCP attempt $retry_count failed, retrying in 2 seconds..."
            sleep 2  # 从3秒减少到2秒
        fi
    done
    
    print_error "All DHCP clients failed after $MAX_DHCP_RETRIES attempts"
    return 1
}

#############################################
# NetworkManager集成模块
# ============================================
# 功能：处理与NetworkManager的冲突
# - 检测NM是否管理接口
# - 临时取消NM管理（避免冲突）
# - 连接完成后恢复NM管理
#############################################

# 检查NetworkManager是否管理指定接口
# 参数：$1 - 网络接口名称
# 返回值：0=NM正在管理，1=NM未管理或未运行
check_nm_manages_interface() {
    local interface="$1"
    
    if [ -z "$interface" ]; then
        return 1
    fi
    
    if ! command -v nmcli &> /dev/null; then
        return 1
    fi
    
    # Check if NetworkManager is running
    if ! systemctl is-active --quiet NetworkManager 2>/dev/null; then
        return 1
    fi
    
    # Check if interface is managed by NetworkManager
    local nm_output nm_state
    nm_output=$(nmcli device show "$interface" 2>/dev/null)
    if [ $? -eq 0 ] && [ -n "$nm_output" ]; then
        nm_state=$(echo "$nm_output" | grep "GENERAL.STATE:" | awk '{print $2}' | head -n1)
        ORIGINAL_NM_STATE="$nm_state"
        
        if [ "$DEBUG" = true ]; then
            print_debug "NetworkManager state for $interface: $nm_state"
        fi
        
        if [ "$nm_state" != "unmanaged" ] && [ "$nm_state" != "unavailable" ]; then
            return 0
        fi
    fi
    
    return 1
}

# 临时取消NetworkManager对接口的管理
# 参数：$1 - 网络接口名称
# 返回值：0=成功，1=失败
# 功能：避免NM与wpa_supplicant冲突
# 策略：最多尝试3次，每次间隔1秒
unmanage_interface_in_nm() {
    local interface="$1"
    local max_attempts=3
    local attempt=1
    
    if [ -z "$interface" ]; then
        return 1
    fi
    
    if check_nm_manages_interface "$interface"; then
        print_info "Temporarily unmanaging $interface in NetworkManager..."
        
        while [ $attempt -le $max_attempts ]; do
            if nmcli device set "$interface" managed no 2>/dev/null; then
                sleep 1
                # Verify the change took effect
                local current_state
                current_state=$(nmcli device show "$interface" 2>/dev/null | grep "GENERAL.STATE:" | awk '{print $2}' | head -n1)
                if [ "$current_state" = "unmanaged" ]; then
                    if [ "$DEBUG" = true ]; then
                        print_debug "Successfully unmanaged $interface (attempt $attempt)"
                    fi
                    return 0
                fi
            fi
            
            attempt=$((attempt + 1))
            if [ $attempt -le $max_attempts ]; then
                if [ "$DEBUG" = true ]; then
                    print_debug "Unmanage attempt $((attempt-1)) failed, retrying..."
                fi
                sleep 1
            fi
        done
        
        print_error "Failed to unmanage $interface after $max_attempts attempts"
        return 1
    fi
    
    return 0  # Interface not managed by NetworkManager, no action needed
}

# 恢复NetworkManager对接口的管理
# 参数：$1 - 网络接口名称
# 返回值：0=成功，1=失败（NM未运行或命令不存在）
# 使用场景：连接完成或失败后恢复NM管理
restore_interface_in_nm() {
    local interface="$1"
    
    if command -v nmcli &> /dev/null && systemctl is-active --quiet NetworkManager 2>/dev/null; then
        print_info "Restoring $interface management in NetworkManager..."
        nmcli device set "$interface" managed yes 2>/dev/null
        return 0
    fi
    
    return 1
}

#############################################
# 连接信息获取模块
# ============================================
# 功能：获取和显示WiFi连接详细信息
# - 信号强度
# - 频率
# - 传输/接收速率
# - 弱信号警告
#############################################

# 获取连接信息（信号强度、频率、速率等）
# 参数：$1 - 网络接口名称
get_connection_info() {
    local interface="$1"
    
    if [ -z "$interface" ]; then
        return 1
    fi
    
    # Check signal strength
    local signal freq bssid ssid tx_bitrate rx_bitrate
    local link_info
    link_info=$(iw dev "$interface" link 2>/dev/null)
    
    if [ -n "$link_info" ]; then
        signal=$(echo "$link_info" | grep "signal:" | awk '{print $2}')
        freq=$(echo "$link_info" | grep "freq:" | awk '{print $2}')
        tx_bitrate=$(echo "$link_info" | grep "tx bitrate:" | awk '{print $3, $4}')
        rx_bitrate=$(echo "$link_info" | grep "rx bitrate:" | awk '{print $3, $4}')
        
        if [ -n "$signal" ]; then
            print_info "Signal strength: ${signal} dBm"
            
            # Warn about weak signal
            local signal_num
            signal_num=$(echo "$signal" | tr -d '-')
            if [ "$signal_num" -gt 70 ]; then
                print_info "Warning: Signal strength is weak (< -70 dBm)"
            fi
        fi
        
        if [ -n "$freq" ]; then
            print_info "Frequency: ${freq} MHz"
        fi
        
        if [ -n "$tx_bitrate" ]; then
            print_info "TX bitrate: $tx_bitrate"
        fi
        
        if [ "$DEBUG" = true ] && [ -n "$rx_bitrate" ]; then
            print_debug "RX bitrate: $rx_bitrate"
        fi
    fi
}


#############################################
# 系统验证模块
# ============================================
# 功能：验证系统是否满足运行要求
# - 检查必需的命令是否存在
# - 检查至少一个DHCP客户端
# - 检查内核无线支持
#############################################

# 验证系统要求
# 返回值：0=所有要求满足，1=缺少依赖
# 必需命令：
#   - ip, iw, wpa_supplicant, wpa_cli
# 可选命令：
#   - wpa_passphrase（有备用方法）
# 必需至少一个DHCP客户端：
#   - dhclient, dhcpcd, 或 udhcpc
validate_system_requirements() {
    local missing_deps=0
    local required_commands=("ip" "iw" "wpa_supplicant" "wpa_cli")
    
    if [ "$DEBUG" = true ]; then
        print_debug "Validating system requirements..."
    fi
    
    # Check required commands
    for cmd in "${required_commands[@]}"; do
        if ! command -v "$cmd" &> /dev/null; then
            print_error "Required command missing: $cmd"
            missing_deps=$((missing_deps + 1))
        fi
    done
    
    # wpa_passphrase is optional (script has fallback to manual config generation)
    if ! command -v "wpa_passphrase" &> /dev/null; then
        if [ "$DEBUG" = true ]; then
            print_debug "wpa_passphrase not found, will use manual config generation if needed"
        fi
    fi
    
    # Check for at least one DHCP client
    local dhcp_found=false
    local dhcp_clients=("dhclient" "dhcpcd" "udhcpc")
    for client in "${dhcp_clients[@]}"; do
        if command -v "$client" &> /dev/null; then
            dhcp_found=true
            break
        fi
    done
    
    if [ "$dhcp_found" = false ]; then
        print_error "No DHCP client found (dhclient, dhcpcd, or udhcpc required)"
        missing_deps=$((missing_deps + 1))
    fi
    
    # Check kernel wireless support
    if [ ! -d "/sys/class/net" ]; then
        print_error "Kernel wireless support not available"
        missing_deps=$((missing_deps + 1))
    fi
    
    if [ $missing_deps -gt 0 ]; then
        print_error "$missing_deps system requirement(s) not met"
        print_info "Install required packages: sudo apt-get install wpasupplicant iw iproute2"
        return 1
    fi
    
    if [ "$DEBUG" = true ]; then
        print_debug "All system requirements validated successfully"
    fi
    return 0
}

# 输入验证和安全模块
# ============================================
# 功能：验证和清理用户输入，防止注入攻击
# - 验证接口名称格式和长度
# - 验证超时值范围
# - 验证WiFi凭据格式
#############################################

# 验证接口名称
# 参数：$1 - 接口名称
# 返回值：0=有效，1=无效
# 规则：
#   - 只允许字母、数字、下划线、连字符、点号
#   - 最大长度15个字符
sanitize_interface() {
    local iface="$1"
    
    if [ -z "$iface" ]; then
        return 1
    fi
    
    # Only allow: letters, numbers, underscore, hyphen, dot
    if [[ ! "$iface" =~ ^[a-zA-Z0-9_.-]+$ ]]; then
        print_error "Invalid interface name: contains illegal characters"
        print_info "Interface name can only contain: letters, numbers, _, -, ."
        return 1
    fi
    
    # Check length (max: 15 characters)
    if [ ${#iface} -gt 15 ]; then
        print_error "Interface name too long (max 15 characters)"
        return 1
    fi
    
    return 0
}

# 验证超时值
# 参数：$1 - 超时值（秒）
# 返回值：0=有效，1=无效
# 规则：
#   - 必须是正整数
#   - 范围：5-300秒
sanitize_timeout() {
    local timeout="$1"
    
    # Check if it's a positive integer
    if ! [[ "$timeout" =~ ^[0-9]+$ ]]; then
        print_error "Invalid timeout value: must be a positive integer"
        return 1
    fi
    
    # Check reasonable range (5-300 seconds)
    if [ "$timeout" -lt 5 ] || [ "$timeout" -gt 300 ]; then
        print_error "Timeout out of range (must be between 5-300 seconds)"
        return 1
    fi
    
    return 0
}

# Note: Injection detection has been removed

# 验证WiFi凭据
# 参数：
#   $1 - SSID
#   $2 - 密码（可为空，表示开放WiFi）
# 返回值：0=有效，1=无效
# 规则：
#   - SSID不能为空
#   - 密码可为空（开放WiFi）
validate_wifi_credentials() {
    local ssid="$1"
    local password="<密码_01>"
    local has_error=false
    
    if [ "$DEBUG" = true ]; then
        print_debug "Validating WiFi credentials..."
        print_debug "SSID: '$ssid' (length: ${#ssid})"
        if [ -z "$password" ]; then
            print_debug "Password: <密码_01> WIFI - No password]"
        else
            print_debug "Password: <密码_01> (length: ${#password})"
        fi
    fi
    
    # Note: Upper layer has already filtered SSID and password
    # We only perform basic length validation here
    
    # Validate SSID - only check if not empty
    if [ -z "$ssid" ]; then
        print_error "SSID cannot be empty"
        has_error=true
    fi
    
    # Password validation - allow any length including empty (for open WiFi)
    # No length restrictions as upper layer handles validation
    if [ -z "$password" ]; then
        # Empty password means open WiFi
        if [ "$DEBUG" = true ]; then
            print_debug "Open WiFi detected (no password required)"
        fi
    fi
    
    if [ "$has_error" = true ]; then
        return 1
    fi
    
    if [ "$DEBUG" = true ]; then
        print_debug "WiFi credentials validation passed"
    fi
    
    return 0
}

#############################################
# 断开连接模块
# ============================================
# 功能：断开WiFi连接并清理相关进程
# - 优雅断开（wpa_cli）
# - 强制停止进程（wpa_supplicant、DHCP客户端）
# - 重置网络接口
# - 恢复NetworkManager管理
#############################################

# 断开单个接口的连接
# 参数：$1 - 网络接口名称
# 断开流程：
#   1. 通过wpa_cli优雅断开
#   2. 停止wpa_supplicant进程
#   3. 通过iw命令断开
#   4. 停止所有DHCP客户端
#   5. 刷新IP并重置接口
#   6. 验证所有进程已停止
#   7. 恢复NetworkManager管理
disconnect_single_interface() {
    local iface="$1"
    
    CURRENT_SSID=$(iw dev "$iface" link 2>/dev/null | grep "SSID" | awk '{print $2}')
    
    if [ -n "$CURRENT_SSID" ]; then
        print_info "Currently connected to: $CURRENT_SSID"
    fi
    
    # Step 1: Disconnect via wpa_cli (graceful)
    if command -v wpa_cli &> /dev/null; then
        if pgrep -f "wpa_supplicant.*$iface" > /dev/null 2>&1; then
            print_info "Disconnecting via wpa_cli..."
            wpa_cli -i "$iface" disconnect 2>/dev/null
            sleep 0.2
            wpa_cli -i "$iface" terminate 2>/dev/null
            sleep 0.3
        fi
    fi
    
    # Step 2: Stop wpa_supplicant processes (with force kill if needed)
    if pgrep -f "wpa_supplicant.*$iface" > /dev/null 2>&1; then
        print_info "Stopping wpa_supplicant processes..."
        stop_process_by_pattern "wpa_supplicant.*$iface" "TERM" 3 true
    fi
    
    # Step 3: Disconnect via iw command
    iw dev "$iface" disconnect 2>/dev/null
    
    # Step 4: Stop all DHCP clients (with force kill if needed)
    print_info "Stopping DHCP clients..."
    local dhcp_patterns=("dhclient.*$iface" "dhcpcd.*$iface" "udhcpc.*$iface")
    for pattern in "${dhcp_patterns[@]}"; do
        stop_process_by_pattern "$pattern" "TERM" 2 true
    done
    
    # Step 5: Flush IP and reset interface
    print_info "Resetting interface..."
    ip addr flush dev "$iface" 2>/dev/null
    ip link set "$iface" down 2>/dev/null
    sleep 0.2
    ip link set "$iface" up 2>/dev/null
    
    # Step 6: Final verification - ensure no related processes are running
    local remaining_processes=0
    if pgrep -f "wpa_supplicant.*$iface" > /dev/null 2>&1; then
        print_error "Warning: wpa_supplicant processes still running for $iface"
        remaining_processes=$((remaining_processes + 1))
    fi
    
    for pattern in "${dhcp_patterns[@]}"; do
        if pgrep -f "$pattern" > /dev/null 2>&1; then
            print_error "Warning: DHCP client processes still running for $iface"
            remaining_processes=$((remaining_processes + 1))
        fi
    done
    
    if [ $remaining_processes -eq 0 ]; then
        if [ "$DEBUG" = true ]; then
            print_debug "All processes stopped successfully"
        fi
    else
        print_error "Some processes may still be running. Attempting final cleanup..."
        # Final aggressive cleanup
        pkill -KILL -f "wpa_supplicant.*$iface" 2>/dev/null
        for pattern in "${dhcp_patterns[@]}"; do
            pkill -KILL -f "$pattern" 2>/dev/null
        done
        sleep 0.3
    fi
    
    # Step 7: Restore NetworkManager management if available
    restore_interface_in_nm "$iface"
    
    print_success "Interface $iface disconnected"
}

# 断开WiFi连接（主函数）
# 参数：$1 - 网络接口名称（为空则断开所有无线接口）
# 功能：
#   - 如果指定接口，断开该接口
#   - 如果未指定接口，断开所有无线接口
#   - 清理临时日志文件（除非使用-k选项）
disconnect_wifi() {
    local interface="$1"
    
    if [ -z "$interface" ]; then
        WIRELESS_INTERFACES=$(iw dev 2>/dev/null | grep Interface | awk '{print $2}')
        
        if [ -z "$WIRELESS_INTERFACES" ]; then
            print_error "No wireless interfaces found"
            exit $EXIT_INTERFACE_ERROR
        fi
        
        for iface in $WIRELESS_INTERFACES; do
            disconnect_single_interface "$iface"
        done
    else
        if ! sanitize_interface "$interface"; then
            exit $EXIT_INPUT_ERROR
        fi
        
        if ! ip link show "$interface" &> /dev/null; then
            print_error "Network interface '$interface' does not exist"
            exit $EXIT_INTERFACE_ERROR
        fi
        
        disconnect_single_interface "$interface"
    fi
    
    # Clean temporary logs and files
    if [ "$KEEP_LOGS" = false ]; then
        # Clean SSID-based temp files (wpa_ssid.log format)
        rm -f "${SCRIPT_DIR}"/wpa_*.conf "${SCRIPT_DIR}"/wpa_*.log "${SCRIPT_DIR}"/wpa_script_*.log "${SCRIPT_DIR}"/wpa_error_*.txt 2>/dev/null
        # Clean old format files (backward compatibility)
        rm -f "${SCRIPT_DIR}"/wpa_supplicant_*.conf "${SCRIPT_DIR}"/wpa_supplicant_*.log 2>/dev/null
        rm -f "${SCRIPT_DIR}"/wpa_log_*.conf 2>/dev/null
    fi
    
    exit $EXIT_SUCCESS
}

#############################################
# 清理函数（退出时调用）
# ============================================
# 功能：脚本退出时执行清理工作
# - 停止wpa_supplicant（仅在错误或断开时）
# - 恢复NetworkManager管理
# - 清理临时错误文件
# - 释放锁
# 注意：成功连接后保持wpa_supplicant运行以维持连接

# 增强的清理函数（带错误处理）
# 功能：在脚本退出时（正常/错误/中断）执行清理
# 清理策略：
#   - 断开模式：停止wpa_supplicant，恢复NM管理
#   - 错误退出：停止wpa_supplicant，恢复NM管理
#   - 成功连接：保持wpa_supplicant运行，不恢复NM管理
cleanup() {
    local exit_code=$?
    
    # Prevent multiple cleanup calls
    if [ "$CLEANUP_PERFORMED" = true ]; then
        exit $exit_code
    fi
    CLEANUP_PERFORMED=true
    
    if [ "$DEBUG" = true ]; then
        print_debug "Cleanup function called with exit code: $exit_code"
    fi
    
    # Only stop wpa_supplicant if we're in disconnect mode or there was an error
    # For successful connections, keep wpa_supplicant running to maintain the connection
    if [ -n "$INTERFACE" ] && [ "$ACTION" = "disconnect" ]; then
        if [ "$DEBUG" = true ]; then
            print_debug "Stopping wpa_supplicant for $INTERFACE (disconnect mode)"
        fi
        pkill -f "wpa_supplicant.*$INTERFACE" 2>/dev/null || true
        
        # Restore NetworkManager management if it was managing the interface
        if [ "$NM_WAS_MANAGING" = true ]; then
            if [ "$DEBUG" = true ]; then
                print_debug "Restoring NetworkManager management for $INTERFACE"
            fi
            restore_interface_in_nm "$INTERFACE" 2>/dev/null || true
        fi
    elif [ -n "$INTERFACE" ] && [ "$exit_code" -ne 0 ]; then
        # Clean up on error
        if [ "$DEBUG" = true ]; then
            print_debug "Cleaning up wpa_supplicant due to error (exit code: $exit_code)"
        fi
        pkill -f "wpa_supplicant.*$INTERFACE" 2>/dev/null || true
        
        # Restore NetworkManager management if it was managing the interface
        if [ "$NM_WAS_MANAGING" = true ]; then
            if [ "$DEBUG" = true ]; then
                print_debug "Restoring NetworkManager management for $INTERFACE"
            fi
            restore_interface_in_nm "$INTERFACE" 2>/dev/null || true
        fi
    elif [ -n "$INTERFACE" ] && [ "$exit_code" -eq 0 ] && [ "$ACTION" = "connect" ]; then
        # Successful connection - keep wpa_supplicant running
        if [ "$DEBUG" = true ]; then
            print_debug "Keeping wpa_supplicant running for persistent connection"
        fi
        # Don't restore NetworkManager to avoid conflicts with our connection
    fi
    
    # Clean up any temporary error files created during this session
    if [ -n "$TEMP_FILES_TO_CLEAN" ]; then
        rm -f $TEMP_FILES_TO_CLEAN 2>/dev/null || true
    fi
    
    # All configuration and log files are preserved for user reference:
    # - Configuration file: $WPA_CONF (contains connection settings)
    # - WPA Log file: $WPA_LOG (contains wpa_supplicant debug info)
    # - Script Log file: $SCRIPT_LOG (contains script execution log)
    
    # Release lock before exiting

#############################################
# 连接状态检查模块
# ============================================
# 功能：检查连接过程中的错误和状态
# - 检查认证错误（密码错误）
# - 检查网络未找到
# - 检查连接状态
#############################################

# 检查AP加密需求与配置是否匹配
# 参数：$1 - 网络接口名称
# 返回值：
#   0 = 匹配（无问题）
#   2 = AP需要加密但未提供密码
check_ap_encryption() {
    local interface="$1"
    
    [ -z "$SSID" ] && return 0
    
    local scan_result=$(wpa_cli -i "$interface" scan_results 2>/dev/null)
    if [ -z "$scan_result" ]; then
        return 0
    fi
    
    local ap_line=$(echo "$scan_result" | grep -F "$SSID" | tail -1)
    if [ -z "$ap_line" ]; then
        return 0
    fi
    
    if echo "$ap_line" | grep -qE '\[(WPA|WPA2|WPA3|PSK|SAE)[^]]*\]'; then
        if [ -z "$PASSWORD" ]; then
            print_error "WRONG PASSWORD: <密码_01> mismatch (AP requires encryption, no password provided)"
            return 2
        fi
    fi
    
    return 0
}

# 检查连接错误
# 参数：$1 - WPA日志文件路径
# 返回值：
#   0 = 无错误
#   2 = 认证错误（密码错误）
#   4 = 网络未找到
check_connection_errors() {
    local log_file="$1"
    
    if [ ! -f "$log_file" ]; then
        return 1
    fi
    
    if grep -q "WPA: 4-Way Handshake failed\|CTRL-EVENT-SSID-TEMP-DISABLED.*auth_failures\|authentication with.*timed out" "$log_file" 2>/dev/null; then
        if [ -z "$PASSWORD" ]; then
            print_error "Authentication failed (connection rejected)"
        else
            print_error "WRONG PASSWORD: <密码_01> failed"
        fi
        return 2
    fi
    
    if grep -q "skip.*privacy mismatch" "$log_file" 2>/dev/null; then
        print_error "WRONG PASSWORD: <密码_01> mismatch (AP requires encryption, no password provided)"
        return 2
    fi
    
    if grep -q "CTRL-EVENT-DISCONNECTED.*reason=\(2\|15\|23\)" "$log_file" 2>/dev/null; then
        if [ -z "$PASSWORD" ]; then
            print_error "Disconnected by access point (connection rejected)"
        else
            print_error "WRONG PASSWORD: <密码_01> by AP"
        fi
        return 2
    fi
    
    if grep -q "No suitable network found\|CTRL-EVENT-SCAN-FAILED" "$log_file" 2>/dev/null; then
        print_error "WiFi network not found: $SSID"
        return 4
    fi
    
    return 0
}

# 检查wpa_supplicant状态
# 参数：$1 - 网络接口名称
# 返回值：
#   0 = COMPLETED（已连接）
#   1 = 其他状态（连接中）
#   2 = DISCONNECTED/INACTIVE（已断开）
check_wpa_status() {
    local interface="$1"
    local status=$(wpa_cli -i "$interface" status 2>/dev/null)
    
    if [ -z "$status" ]; then
        return 1
    fi
    
    local wpa_state=$(echo "$status" | grep "wpa_state=" | cut -d'=' -f2)
    
    if [ "$DEBUG" = true ]; then
        print_debug "State: $wpa_state"
    fi
    
    case "$wpa_state" in
        "COMPLETED") return 0 ;;
        "DISCONNECTED"|"INACTIVE") return 2 ;;
        *) return 1 ;;
    esac
}

#############################################
# 连接流程模块
# ============================================
# 功能：执行完整的WiFi连接流程
# 连接步骤：
#   1. 处理NetworkManager冲突
#   2. 停止现有的wpa_supplicant和DHCP客户端
#   3. 启动网络接口
#   4. 生成WPA配置文件
#   5. 启动wpa_supplicant
#   6. 等待连接建立
#   7. 获取DHCP IP地址
#   8. 显示连接信息
#############################################

# 处理NetworkManager冲突
if check_nm_manages_interface "$INTERFACE"; then
    NM_WAS_MANAGING=true
    if ! unmanage_interface_in_nm "$INTERFACE"; then
        print_info "Failed to unmanage interface in NetworkManager, stopping NetworkManager..."
        systemctl stop NetworkManager 2>/dev/null
        wait_for_process_stop "NetworkManager" 3
    fi
fi

# Stop wpa_supplicant service if running
if systemctl is-active --quiet wpa_supplicant 2>/dev/null; then
    print_info "Stopping wpa_supplicant service..."
    systemctl stop wpa_supplicant 2>/dev/null
    wait_for_process_stop "wpa_supplicant" 3
fi

# Stop existing wpa_supplicant for this interface (with force kill)
if pgrep -f "wpa_supplicant.*$INTERFACE" > /dev/null 2>&1; then
    print_info "Stopping existing wpa_supplicant for $INTERFACE..."
    # Try graceful termination first
    if command -v wpa_cli &> /dev/null; then
        wpa_cli -i "$INTERFACE" disconnect 2>/dev/null
        wpa_cli -i "$INTERFACE" terminate 2>/dev/null
        sleep 0.3
    fi
    # Force stop with kill if needed
    stop_process_by_pattern "wpa_supplicant.*$INTERFACE" "TERM" 3 true
fi

# Stop any DHCP clients for this interface
print_info "Cleaning up any existing DHCP clients for $INTERFACE..."
dhcp_patterns=("dhclient.*$INTERFACE" "dhcpcd.*$INTERFACE" "udhcpc.*$INTERFACE")
for pattern in "${dhcp_patterns[@]}"; do
    if pgrep -f "$pattern" > /dev/null 2>&1; then
        stop_process_by_pattern "$pattern" "TERM" 2 true
    fi
done

# Bring up interface
print_info "Bringing up interface $INTERFACE..."
ip link set "$INTERFACE" down 2>/dev/null
sleep 0.05  # 从0.1秒减少到0.05秒
if ! ip link set "$INTERFACE" up; then
    print_error "Failed to bring up interface"
    exit $EXIT_INTERFACE_ERROR
fi

# Brief wait for interface to be ready
sleep $INTERFACE_UP_WAIT  # 使用配置常量

# Generate configuration (skip for reconnect mode)
if [ "$USE_LAST_CONF" = false ]; then
    print_info "Generating configuration..."

    # Check if this is an open WiFi (no password)
    if [ -z "$PASSWORD" ]; then
        print_info "Configuring for open WiFi (no encryption)..."
        if ! generate_open_config "$SSID" "$WPA_CONF"; then
            print_error "Failed to generate open WiFi configuration"
            exit $EXIT_CONFIG_ERROR
        fi
    else
        if ! generate_wpa_config "$SSID" "$PASSWORD" "$WPA_CONF"; then
            print_error "Failed to generate WPA configuration"
            exit $EXIT_CONFIG_ERROR
        fi
    fi

    if [ "$DEBUG" = true ]; then
        print_debug "Configuration file content:"
        if [ -z "$PASSWORD" ]; then
            cat "$WPA_CONF"
        else
            cat "$WPA_CONF" | grep -v "psk="
        fi
    fi
else
    print_info "Using existing configuration file: $WPA_CONF"
    if [ "$DEBUG" = true ]; then
        print_debug "Configuration file content:"
        cat "$WPA_CONF" | grep -v "psk="
    fi
fi

# Start wpa_supplicant
print_info "Connecting to $SSID..."

wpa_supplicant -B -i "$INTERFACE" -c "$WPA_CONF" -f "$WPA_LOG" -d

if [ $? -ne 0 ]; then
    print_error "Failed to start wpa_supplicant"
    exit $EXIT_CONFIG_ERROR
fi

# Brief wait for wpa_supplicant to initialize
sleep $WPA_START_WAIT  # 从0.5秒减少到0.3秒

# Wait for connection
print_info "Waiting for connection (timeout: ${TIMEOUT}s)..."
COUNTER=0
CONNECTED=false
NOT_FOUND_COUNTER=0
NOT_FOUND_GRACE=10  # seconds to keep retrying after "network not found" (handles driver reload)

while [ $COUNTER -lt $TIMEOUT ]; do
    check_connection_errors "$WPA_LOG"
    ERROR_CODE=$?

    if [ $ERROR_CODE -eq 2 ]; then
        exit $EXIT_AUTH_ERROR
    elif [ $ERROR_CODE -eq 3 ]; then
        exit $EXIT_AUTH_ERROR
    elif [ $ERROR_CODE -eq 4 ]; then
        NOT_FOUND_COUNTER=$((NOT_FOUND_COUNTER + CHECK_INTERVAL))
        if [ $NOT_FOUND_COUNTER -ge $NOT_FOUND_GRACE ]; then
            exit $EXIT_NETWORK_NOT_FOUND
        fi
        # Grace period: let wpa_supplicant retry scanning (e.g. right after driver reload)
    else
        NOT_FOUND_COUNTER=0
    fi
    
    check_ap_encryption "$INTERFACE"
    ENCRYPT_CODE=$?
    
    if [ $ENCRYPT_CODE -eq 2 ]; then
        exit $EXIT_AUTH_ERROR
    fi
    
    check_wpa_status "$INTERFACE"
    STATUS_CODE=$?
    
    if [ $STATUS_CODE -eq 0 ]; then
        CONNECTED=true
        break
    elif [ $STATUS_CODE -eq 2 ]; then
        print_error "Connection failed"
        check_connection_errors "$WPA_LOG"
        exit $EXIT_INTERFACE_ERROR
    fi
    
    if [ "$QUIET" != true ]; then
        if [ "$DEBUG" = true ]; then
            CURRENT_STATE=$(wpa_cli -i "$INTERFACE" status 2>/dev/null | grep "wpa_state=" | cut -d'=' -f2)
            echo -n "[$CURRENT_STATE]"
        else
            echo -n "."
        fi
    fi
    
    # 使用更短的间隔进行状态检查（从2秒改为1秒）
    sleep $CHECK_INTERVAL
    COUNTER=$((COUNTER + CHECK_INTERVAL))
done

if [ "$QUIET" != true ]; then
    echo ""
fi

if [ "$CONNECTED" = false ]; then
    print_error "Connection timeout"
    
    if [ "$DEBUG" = true ]; then
        print_debug "Log content:"
        tail -30 "$WPA_LOG"
    fi
    
    exit $EXIT_CONNECTION_TIMEOUT
fi

print_success "WiFi connected!"

# Get IP address
print_info "Obtaining IP address..."

if ! get_dhcp_address "$INTERFACE"; then
    exit $EXIT_DHCP_ERROR
fi

IP_ADDR=$(ip addr show "$INTERFACE" | grep "inet " | awk '{print $2}')

if [ -z "$IP_ADDR" ]; then
    print_error "Failed to obtain IP address"
    exit $EXIT_DHCP_ERROR
fi

# Get comprehensive connection info
get_connection_info "$INTERFACE"

# Display info
if [ "$QUIET" != true ]; then
    print_success "========== Connection Info =========="
    wpa_cli -i "$INTERFACE" status 2>/dev/null | grep -E "ssid=|bssid=|freq=|key_mgmt="
    echo -e "${GREEN}IP Address:${NC} $IP_ADDR"
    
    # Check if it's an open WiFi by examining key_mgmt (not just PASSWORD variable)
    KEY_MGMT=$(wpa_cli -i "$INTERFACE" status 2>/dev/null | grep "key_mgmt=" | cut -d'=' -f2)
    if [ "$KEY_MGMT" = "NONE" ]; then
        echo -e "${YELLOW}Security:${NC} Open (No encryption)"
    fi
    print_success "====================================="
fi

print_success "WiFi connection completed!"

# Log success if keeping logs
if [ "$KEEP_LOGS" = true ]; then
    log_message "Connected to $SSID - IP: $IP_ADDR"
fi

if [ "$QUIET" != true ]; then
    print_success "WiFi connection established and will remain active"
    print_info "wpa_supplicant is running to maintain the connection"
    print_info "To disconnect: sudo $0 -i $INTERFACE -S"

    # Keep all logs after successful connection for user reference
    print_info "Connection logs preserved:"
    print_info "  WPA Log: $WPA_LOG"
    print_info "  Script Log: $SCRIPT_LOG"

    if [ "$KEEP_LOGS" = true ] && [ -n "$PERSISTENT_LOG" ] && [ -n "$PERSISTENT_SCRIPT_LOG" ]; then
        # Copy temp logs to persistent location if keeping logs
        if cp "$WPA_LOG" "$PERSISTENT_LOG" 2>/dev/null; then
            print_info "WPA log also saved to: $PERSISTENT_LOG"
        fi
        if cp "$SCRIPT_LOG" "$PERSISTENT_SCRIPT_LOG" 2>/dev/null; then
            print_info "Script log also saved to: $PERSISTENT_SCRIPT_LOG"
        fi
    fi
else
    # In quiet mode, still copy logs if keeping them, just don't show messages
    if [ "$KEEP_LOGS" = true ] && [ -n "$PERSISTENT_LOG" ] && [ -n "$PERSISTENT_SCRIPT_LOG" ]; then
        cp "$WPA_LOG" "$PERSISTENT_LOG" 2>/dev/null
        cp "$SCRIPT_LOG" "$PERSISTENT_SCRIPT_LOG" 2>/dev/null
    fi
fi

# NetworkManager will be restored automatically by cleanup function on exit
exit $EXIT_SUCCESS

# 错误退出（支持自定义退出码）
error_exit() {
    local exit_code="$EXIT_GENERAL_ERROR"
    
    # 如果第一个参数是数字，则作为退出码
    if [[ "$1" =~ ^[0-9]+$ ]]; then
        exit_code="$1"
        shift
    fi
    
    local msg="$*"
    local code_desc=$(get_exit_code_message $exit_code)
    
    echo ""
    echo -e "${RED}╔════════════════════════════════════════╗${NC}" >&2
    echo -e "${RED}║          AP 启动失败                   ║${NC}" >&2
    echo -e "${RED}╚════════════════════════════════════════╝${NC}" >&2
    echo ""
    echo -e "${RED}错误码: ${YELLOW}$exit_code${NC}" >&2
    echo -e "${RED}错误类型: ${YELLOW}$code_desc${NC}" >&2
    echo -e "${RED}错误详情: ${NC}$msg" >&2
    echo ""
    echo -e "${BLUE}提示:${NC}" >&2
    echo -e "  - 查看详细日志: cat $ERROR_LOG" >&2
    echo -e "  - 查看 hostapd 日志: cat $HOSTAPD_LOG" >&2
    echo -e "  - 查看 dnsmasq 日志: cat $DNSMASQ_LOG" >&2
    echo ""
    
    # 记录到错误日志
    if [ -d "$LOG_DIR" ]; then
        echo "[$(date '+%Y-%m-%d %H:%M:%S')] [ERROR] 退出码: $exit_code | 类型: $code_desc | 详情: $msg" >> "$ERROR_LOG"
    fi
    
    exit "$exit_code"
}

# ==================== 锁机制 ====================

acquire_lock() {
    local timeout=10
    local waited=0
    
    log_info "尝试获取锁..."
    
    while [ $waited -lt $timeout ]; do
        if mkdir "$LOCK_FILE" 2>/dev/null; then
            trap 'release_lock' EXIT INT TERM
            echo $$ > "$LOCK_FILE/pid"
            log_info "锁获取成功 (PID: $$)"
            return 0
        fi
        
        # 检查锁是否过期
        if [ -f "$LOCK_FILE/pid" ]; then
            local lock_pid=$(cat "$LOCK_FILE/pid")
            if ! kill -0 "$lock_pid" 2>/dev/null; then
                warning "检测到过期锁 (PID: $lock_pid)，清理中..."
                rm -rf "$LOCK_FILE"
                continue
            else
                warning "检测到另一个实例正在运行 (PID: $lock_pid)"
            fi
        fi
        
        sleep 1
        waited=$((waited + 1))
    done
    
    error_exit "$EXIT_LOCK_CONFLICT" "无法获取锁，另一个实例正在运行。请等待其完成或使用 'ps aux | grep ap_connect' 检查"
}

release_lock() {
    if [ -d "$LOCK_FILE" ]; then
        rm -rf "$LOCK_FILE"
        log_info "锁已释放"
    fi
}

# 检查网卡是否支持指定频段
check_band_support() {
    local interface="$1"
    local band="$2"
    
    if ! ip link show "$interface" &>/dev/null; then
        error_exit "$EXIT_INTERFACE_ERROR" "网卡 $interface 不存在"
    fi
    
    # 检查网卡能力
    local phy=$(iw dev "$interface" info | grep wiphy | awk '{print $2}')
    if [ -z "$phy" ]; then
        warning "无法检测网卡能力，将尝试继续..."
        return 0
    fi
    
    local capabilities=$(iw phy phy"$phy" info 2>/dev/null || echo "")
    
    if [ "$band" = "5" ]; then
        if ! echo "$capabilities" | grep -q "5[0-9][0-9][0-9] MHz"; then
            error_exit "$EXIT_BAND_NOT_SUPPORTED" "网卡 $interface 不支持 5GHz 频段"
        fi
    fi
}

# 验证信道号是否合法
validate_channel() {
    local band="$1"
    local channel="$2"
    
    # 信道 0 表示自动信道选择 (ACS)，总是合法的
    if [ "$channel" -eq 0 ]; then
        return 0
    fi
    
    if [ "$band" = "2.4" ]; then
        if [ "$channel" -lt 1 ] || [ "$channel" -gt 14 ]; then
            error_exit "$EXIT_CHANNEL_INVALID" "2.4GHz 信道范围: 1-14 (或使用 0 启用自动选择)"
        fi
    elif [ "$band" = "5" ]; then
        # 5GHz 合法信道列表
        local valid_channels="36 40 44 48 52 56 60 64 100 104 108 112 116 120 124 128 132 136 140 144 149 153 157 161 165"
        if ! echo "$valid_channels" | grep -qw "$channel"; then
            error_exit "$EXIT_CHANNEL_INVALID" "5GHz 信道无效。合法信道: $valid_channels (或使用 0 启用自动选择)"
        fi
    fi
}

    if [ "$band" = "5" ]; then
        echo "a"
    else
        echo "g"
    fi
}

# 快速清理进程的辅助函数
quick_kill_processes() {
    local process_name="$1"
    local pids="$2"
    
    if [ -z "$pids" ]; then
        return 0
    fi
    
    # 批量发送 TERM 信号
    for pid in $pids; do
        kill -TERM "$pid" 2>/dev/null || true
    done
    
    # 快速轮询检查（最多等待1秒，每0.1秒检查一次）
    local max_wait=10  # 10 * 0.1 = 1秒
    local wait_count=0
    local all_dead=false
    
    while [ $wait_count -lt $max_wait ]; do
        all_dead=true
        for pid in $pids; do
            if kill -0 "$pid" 2>/dev/null; then
                all_dead=false
                break
            fi
        done
        
        if [ "$all_dead" = true ]; then
            break
        fi
        
        sleep 0.1
        wait_count=$((wait_count + 1))
    done
    
    # 强制终止仍在运行的进程
    for pid in $pids; do
        if kill -0 "$pid" 2>/dev/null; then
            kill -KILL "$pid" 2>/dev/null || true
        fi
    done
}

# 停止系统 DHCP 服务
stop_system_dhcp_services() {
    log_info "检查并停止系统 DHCP 服务..."

    local dhcp_services="udhcpcd"
    local stopped=false

    for svc in $dhcp_services; do
        # 检查 systemctl
        if command -v systemctl &>/dev/null; then
            if systemctl is-active --quiet "$svc" 2>/dev/null; then
                info "停止系统 DHCP 服务: $svc"
                systemctl stop "$svc" 2>/dev/null || true
                systemctl disable "$svc" 2>/dev/null || true
                stopped=true
            fi
        fi

        # 检查 service 命令
        if command -v service &>/dev/null; then
            if service "$svc" status &>/dev/null 2>&1; then
                info "停止系统 DHCP 服务: $svc"
                service "$svc" stop 2>/dev/null || true
                stopped=true
            fi
        fi

        # 检查 udhcpd 进程
        local pids=$(ps aux | grep "[u]dhcpd" | awk '{print $2}')
        if [ -n "$pids" ]; then
            info "停止 udhcpd 进程: $pids"
            kill -TERM $pids 2>/dev/null || true
            stopped=true
        fi
    done

    # 释放占用的网口 IP（清理之前 DHCP 客户端获取的 IP）
    for iface in $(ip -o link show | awk -F': ' '{print $2}'); do
        if ip addr show "$iface" | grep -q "inet .* dhcp"; then
            info "清理 $iface 的 DHCP 地址"
            ip addr flush dev "$iface" 2>/dev/null || true
        fi
    done

    [ "$stopped" = true ] && success "✓ 系统 DHCP 服务已停止" || info "无需停止系统 DHCP 服务"
}

# 检查并清理残留的进程（优化版：同时处理 hostapd 和 dnsmasq）
check_and_kill_existing_processes() {

        success "✓ 残留进程已清理"
    fi
}

# 停止进程
stop_process() {
    local pid_file="$1"
    local process_name="$2"
    
    if [ ! -f "$pid_file" ]; then
        log_info "$process_name 未运行（PID 文件不存在）"
        return 0
    fi
    
    local pid=$(cat "$pid_file")
    
    if ! kill -0 "$pid" 2>/dev/null; then
        log_info "$process_name 进程不存在 (PID: $pid)，清理 PID 文件"
        rm -f "$pid_file"
        return 0
    fi
    
    info "停止 $process_name (PID: $pid)..."
    kill -TERM "$pid" 2>/dev/null || true
    
    # 等待进程结束
    local waited=0
    while kill -0 "$pid" 2>/dev/null && [ $waited -lt 10 ]; do
        sleep 1
        waited=$((waited + 1))
    done
    
    if kill -0 "$pid" 2>/dev/null; then
        warning "$process_name 未响应 SIGTERM，发送 SIGKILL..."
        kill -KILL "$pid" 2>/dev/null || true
        sleep 1
    fi
    
    rm -f "$pid_file"
    success "✓ $process_name 已停止"
}

# 配置网络接口
configure_interface() {
    local interface="$1"
    local ip="$2"
    
    info "配置网络接口 $interface..."
    
    # 关闭接口
    if ! ip link set "$interface" down 2>&1 | tee -a "$LOG_FILE"; then
        error_exit "$EXIT_NETWORK_CONFIG_FAILED" "无法关闭接口 $interface"
    fi
    
    # 清除已有 IP
    ip addr flush dev "$interface" 2>&1 | tee -a "$LOG_FILE" || true
    
    # 启动接口
    if ! ip link set "$interface" up 2>&1 | tee -a "$LOG_FILE"; then
        error_exit "$EXIT_NETWORK_CONFIG_FAILED" "无法启动接口 $interface"
    fi
    
    # 设置 IP
    if ! ip addr add "$ip/24" dev "$interface" 2>&1 | tee -a "$LOG_FILE"; then
        error_exit "$EXIT_NETWORK_CONFIG_FAILED" "无法设置 IP 地址 $ip 到接口 $interface"
    fi
    
    success "✓ 网络接口 $interface 配置完成 (IP: $ip)"
    log_info "网络接口 $interface 配置完成 (IP: $ip)"
}

# 清理网络接口
cleanup_interface() {
    local interface="$1"
    
    info "清理网络接口 $interface..."
    ip addr flush dev "$interface" 2>&1 | tee -a "$LOG_FILE" || true
    ip link set "$interface" down 2>&1 | tee -a "$LOG_FILE" || true
    log_info "网络接口 $interface 清理完成"
}

# 从 hostapd 和 dnsmasq 配置文件加载上次的配置
load_last_config() {
    if [ ! -f "$HOSTAPD_CONF" ]; then
        error_exit "$EXIT_CONFIG_ERROR" "未找到 hostapd 配置文件。请先运行一次 AP 后再使用 -r 选项"
    fi
    
    if [ ! -f "$DNSMASQ_CONF" ]; then
        error_exit "$EXIT_CONFIG_ERROR" "未找到 dnsmasq 配置文件。请先运行一次 AP 后再使用 -r 选项"
    fi
    
    # 从 hostapd.conf 读取配置
    INTERFACE=$(grep "^interface=" "$HOSTAPD_CONF" | cut -d= -f2)
    SSID=$(grep "^ssid=" "$HOSTAPD_CONF" | cut -d= -f2)
    PASSWORD=<密码_01> "^wpa_passphrase=" "$HOSTAPD_CONF" | cut -d= -f2)
    CHANNEL=$(grep "^channel=" "$HOSTAPD_CONF" | cut -d= -f2)
    local hw_mode=$(grep "^hw_mode=" "$HOSTAPD_CONF" | cut -d= -f2)
    
    # 根据 hw_mode 判断频段
    if [ "$hw_mode" = "a" ]; then
        BAND="5"
    else
        BAND="2.4"
    fi
    
    # 从 dnsmasq.conf 读取 DHCP 配置
    local dhcp_range=$(grep "^dhcp-range=" "$DNSMASQ_CONF" | cut -d= -f2)
    DHCP_START=$(echo "$dhcp_range" | cut -d, -f1)
    DHCP_END=$(echo "$dhcp_range" | cut -d, -f2)
    
    # 计算 IP 地址（从 DHCP 起始地址推算网关）
    # 例如: 192.168.100.10 -> 192.168.100.1
    IP=$(echo "$DHCP_START" | sed 's/\.[0-9]*$/\.1/')
    
    # 验证读取的配置
    if [ -z "$INTERFACE" ] || [ -z "$SSID" ] || [ -z "$PASSWORD" ] || [ -z "$CHANNEL" ]; then
        error_exit "$EXIT_CONFIG_ERROR" "配置文件解析失败，请检查 hostapd.conf 和 dnsmasq.conf"
    fi
    
    info "已从配置文件加载上次的配置："
    
    # 信道显示
    local channel_display="$CHANNEL"
    if [ "$CHANNEL" -eq 0 ]; then
        channel_display="自动选择 (ACS)"
    fi
    
    echo "  接口: $INTERFACE"
    echo "  SSID: $SSID"
    echo "  频段: $BAND GHz"
    echo "  信道: $channel_display"
    echo "  IP: $IP"
    echo ""
}

# 解析命令行参数
parse_arguments() {
    INTERFACE="$DEFAULT_INTERFACE"
    SSID="$DEFAULT_SSID"
    PASSWORD="<密码_01>"
    BAND="$DEFAULT_BAND"
    CHANNEL=""
    IP="$DEFAULT_IP"
    DHCP_START="$DEFAULT_DHCP_START"
    DHCP_END="$DEFAULT_DHCP_END"
    DRIVER_NAME="$DEFAULT_DRIVER_NAME"
    RESUME_MODE=false
    STOP_MODE=false
    HOSTAPD_DEBUG_LEVEL=""

    # 检查是否使用 stop 模式
    for arg in "$@"; do
        if [ "$arg" = "-S" ] || [ "$arg" = "--stop" ]; then
            STOP_MODE=true
            break
        fi
    done
    
    # 如果是 stop 模式，先解析接口参数
    if [ "$STOP_MODE" = true ]; then
        # 检查是否显式指定了 -i 参数
        local has_interface_flag=false
        local interface_value=""
        local temp_args=("$@")
        local i=0
        while [ $i -lt ${#temp_args[@]} ]; do
            if [ "${temp_args[$i]}" = "-i" ] || [ "${temp_args[$i]}" = "--interface" ]; then
                has_interface_flag=true
                if [ $((i+1)) -lt ${#temp_args[@]} ]; then
                    interface_value="${temp_args[$((i+1))]}"
                    INTERFACE="$interface_value"
                fi
                break
            fi
            i=$((i+1))
        done
        
        # 在 stop 模式下，必须显式指定接口
        if [ "$has_interface_flag" = false ] || [ -z "$interface_value" ]; then
            error_exit "$EXIT_PARAM_INVALID" "停止 AP 时必须指定接口，请使用: $0 -i <接口> -S"
        fi
        
        # 执行停止并退出
        stop_ap "$INTERFACE"
        local stop_exit_code=$?
        exit "$stop_exit_code"
    fi
    
    # 检查是否使用 resume 模式
    for arg in "$@"; do
        if [ "$arg" = "-r" ] || [ "$arg" = "--resume" ]; then
            RESUME_MODE=true
            break
        fi
    done
    
    # 如果是 resume 模式，加载上次配置并返回
    if [ "$RESUME_MODE" = true ]; then
        load_last_config
        return 0
    fi
    
    while [[ $# -gt 0 ]]; do
        case $1 in
            -i|--interface)
                INTERFACE="$2"
                shift 2
                ;;
            -s|--ssid)
                SSID="$2"
                shift 2
                ;;
            -p|--password)
                PASSWORD="<密码_01>"
                shift 2
                ;;
            -b|--band)
                BAND="$2"
                shift 2
                ;;
            -c|--channel)
                CHANNEL="$2"
                shift 2
                ;;
            --ip)
                IP="$2"
                shift 2
                ;;
            --dhcp-start)
                DHCP_START="$2"
                shift 2
                ;;
            --dhcp-end)
                DHCP_END="$2"
                shift 2
                ;;
            -d|--driver)
                DRIVER_NAME="$2"
                shift 2
                ;;
            -r|--resume)
                # 已在前面处理
                shift
                ;;
            -hd)
                case "$2" in
                    d|dd|ddd)
                        HOSTAPD_DEBUG_LEVEL="$2"
                        ;;
                    *)
                        error_exit "$EXIT_PARAM_INVALID" "hostapd 调试级别无效: $2 (有效值: d, dd, ddd)"
                        ;;
                esac
                shift 2
                ;;
            -S|--stop)
                # 已在前面处理
                shift
                ;;
    --exit-codes|--exitcodes)
                show_exit_codes
                exit 0
                ;;
            --last-error|--lasterror)
                show_last_error
                exit 0
                ;;
            -h|--help)
                show_help
                exit 0
                ;;
            *)
                error_exit "$EXIT_PARAM_INVALID" "未知选项: $1\n使用 -h 或 --help 查看帮助"
                ;;
        esac
    done
    
    # 参数验证
    if [ ${#PASSWORD} -lt 8 ]; then
        error_exit "$EXIT_PARAM_INVALID" "密码长度至少为 8 个字符"
    fi
    
    # 标准化频段格式
    if [ "$BAND" = "2.4" ] || [ "$BAND" = "2" ] || [ "$BAND" = "24" ]; then
        BAND="2.4"
    elif [ "$BAND" = "5" ] || [ "$BAND" = "5.0" ]; then
        BAND="5"
    else
        error_exit "$EXIT_PARAM_INVALID" "频段必须是 2.4 或 5"
    fi
    
    # 如果没有指定信道，使用默认值
    if [ -z "$CHANNEL" ]; then
        if [ "$BAND" = "5" ]; then
            CHANNEL="$DEFAULT_CHANNEL_5"
        else
            CHANNEL="$DEFAULT_CHANNEL_24"
        fi
    fi
    
    # 验证信道
    validate_channel "$BAND" "$CHANNEL"
}

# 生成 hostapd 配置
generate_hostapd_config() {
    local config_file="$SCRIPT_DIR/runtime/hostapd.conf"
    local hw_mode=$(get_hw_mode "$BAND")
    
    mkdir -p "$SCRIPT_DIR/runtime"
    
    # 判断是否使用自动信道选择（ACS）
    local use_acs=0
    if [ "$CHANNEL" -eq 0 ]; then
        use_acs=1
        info "启用自动信道选择 (ACS)"
    fi
    
    cat > "$config_file" <<EOF
# Hostapd 配置文件
# 由 ap_connect.sh 自动生成于 $(date)

interface=$INTERFACE
ssid=$SSID
hw_mode=$hw_mode
driver=nl80211

# 基本配置
#dtim_period=1
#macaddr_acl=0
wmm_enabled=1
#auth_algs=1
#ignore_broadcast_ssid=0
#beacon_int=100
country_code=CN

# 安全配置
wpa=2
wpa_passphrase=$PASSWORD
wpa_key_mgmt=WPA-PSK
wpa_pairwise=CCMP
rsn_pairwise=CCMP

# 802.11n 配置
ieee80211n=1
ht_capab=[SHORT-GI-20][HT40+][SHORT-GI-40]
EOF

    # 信道配置：ACS 或固定信道
    if [ $use_acs -eq 1 ]; then
        cat >> "$config_file" <<EOF

# 自动信道选择 (ACS) 配置
channel=0
acs_num_scans=5
acs_chan_bias=1:0.8 6:0.8 11:0.8
EOF
        info "已配置 ACS：将扫描并选择最佳信道"
    else
        cat >> "$config_file" <<EOF

# 固定信道配置
channel=$CHANNEL
EOF
        
        # 5GHz 特定配置
        if [ "$BAND" = "5" ]; then
            cat >> "$config_file" <<EOF
ieee80211ac=1
vht_capab=[SHORT-GI-20][SHORT-GI-40][VHT40+]
#vht_oper_centr_freq_seg0_idx=$((CHANNEL + 6))
EOF
        fi
    fi

    # 日志配置
    cat >> "$config_file" <<EOF

# 日志配置
logger_syslog=-1
logger_syslog_level=2
logger_stdout=-1
logger_stdout_level=2
EOF
}

# 停止 AP
stop_ap() {
    local target_interface="$1"
    
    if [ -z "$target_interface" ]; then
        error_exit "$EXIT_PARAM_INVALID" "停止 AP 时必须指定接口，请使用: $0 -i <接口> -S"
    fi
    
    echo ""
    info "========== 强制停止 AP (接口: $target_interface) =========="
    echo ""
    
    local stopped_any=false
    
    # 检查并停止指定接口的 hostapd 进程
    info "检查接口 $target_interface 的 hostapd 进程..."
    local target_hostapd_pids=""
    local all_hostapd_pids=$(pgrep -x hostapd 2>/dev/null || true)
    
    # 遍历所有 hostapd 进程，找到使用指定接口的进程
    for pid in $all_hostapd_pids; do
        # 检查进程的命令行参数，找到配置文件
        local conf_file=$(ps -p "$pid" -o args= 2>/dev/null | grep -oE '[^ ]+\.conf' | head -1 || true)
        if [ -n "$conf_file" ] && [ -f "$conf_file" ]; then
            local interface=$(grep "^interface=" "$conf_file" 2>/dev/null | cut -d= -f2)
            if [ "$interface" = "$target_interface" ]; then
                target_hostapd_pids="$target_hostapd_pids $pid"
            fi
        fi
    done
    
    # 检查并停止指定接口的 dnsmasq 进程
    info "检查接口 $target_interface 的 dnsmasq 进程..."
    local target_dnsmasq_pids=""
    local all_dnsmasq_pids=$(pgrep -x dnsmasq 2>/dev/null || true)
    
    # 遍历所有 dnsmasq 进程，找到使用指定接口的进程
    for pid in $all_dnsmasq_pids; do
        # 检查进程的命令行参数，找到配置文件
        local conf_file=$(ps -p "$pid" -o args= 2>/dev/null | grep -oE '[^ ]+\.conf' | head -1 || true)
        if [ -n "$conf_file" ] && [ -f "$conf_file" ]; then
            local interface=$(grep "^interface=" "$conf_file" 2>/dev/null | cut -d= -f2)
            if [ "$interface" = "$target_interface" ]; then
                target_dnsmasq_pids="$target_dnsmasq_pids $pid"
            fi
        fi
    done
    
    # 如果找到了相关进程，停止它们
    if [ -n "$target_hostapd_pids" ]; then
        warning "发现接口 $target_interface 的 hostapd 进程: $target_hostapd_pids"
        quick_kill_processes "hostapd" "$target_hostapd_pids"
        stopped_any=true
    fi
    
    if [ -n "$target_dnsmasq_pids" ]; then
        warning "发现接口 $target_interface 的 dnsmasq 进程: $target_dnsmasq_pids"
        quick_kill_processes "dnsmasq" "$target_dnsmasq_pids"
        stopped_any=true
    fi
    
    # 清理指定接口的 PID 文件（如果配置文件存在且匹配）
    if [ -f "$HOSTAPD_CONF" ]; then
        local conf_interface=$(grep "^interface=" "$HOSTAPD_CONF" 2>/dev/null | cut -d= -f2)
        if [ "$conf_interface" = "$target_interface" ]; then
            [ -f "$PID_FILE" ] && rm -f "$PID_FILE" && info "清理 hostapd PID 文件"
        fi
    fi
    
    if [ -f "$DNSMASQ_CONF" ]; then
        local conf_interface=$(grep "^interface=" "$DNSMASQ_CONF" 2>/dev/null | cut -d= -f2)
        if [ "$conf_interface" = "$target_interface" ]; then
            [ -f "$DNSMASQ_PID" ] && rm -f "$DNSMASQ_PID" && info "清理 dnsmasq PID 文件"
        fi
    fi
    
    # 清理指定接口的网络配置
    if ip link show "$target_interface" &>/dev/null; then
        cleanup_interface "$target_interface"
        stopped_any=true
    fi
    
    # 清理锁文件（如果配置文件存在且匹配）
    if [ -f "$HOSTAPD_CONF" ]; then
        local conf_interface=$(grep "^interface=" "$HOSTAPD_CONF" 2>/dev/null | cut -d= -f2)
        if [ "$conf_interface" = "$target_interface" ] && [ -d "$LOCK_FILE" ]; then
            local lock_pid=$(cat "$LOCK_FILE/pid" 2>/dev/null || echo "")
            warning "清理锁文件 (之前的 PID: $lock_pid)"
            rm -rf "$LOCK_FILE"
            log_info "锁文件已强制清理"
            stopped_any=true
        fi
    fi
    
    # 额外检查：确保指定接口没有残留进程
    local remaining_hostapd=0
    local remaining_dnsmasq=0
    for pid in $(pgrep -x hostapd 2>/dev/null || true); do
        local conf_file=$(ps -p "$pid" -o args= 2>/dev/null | grep -oE '[^ ]+\.conf' | head -1 || true)
        if [ -n "$conf_file" ] && [ -f "$conf_file" ]; then
            local interface=$(grep "^interface=" "$conf_file" 2>/dev/null | cut -d= -f2)
            if [ "$interface" = "$target_interface" ]; then
                remaining_hostapd=$((remaining_hostapd + 1))
            fi
        fi
    done
    
    for pid in $(pgrep -x dnsmasq 2>/dev/null || true); do
        local conf_file=$(ps -p "$pid" -o args= 2>/dev/null | grep -oE '[^ ]+\.conf' | head -1 || true)
        if [ -n "$conf_file" ] && [ -f "$conf_file" ]; then
            local interface=$(grep "^interface=" "$conf_file" 2>/dev/null | cut -d= -f2)
            if [ "$interface" = "$target_interface" ]; then
                remaining_dnsmasq=$((remaining_dnsmasq + 1))
            fi
        fi
    done

    if [ "$remaining_hostapd" -gt 0 ] || [ "$remaining_dnsmasq" -gt 0 ]; then
        warning "检测到接口 $target_interface 的残留进程: hostapd=$remaining_hostapd, dnsmasq=$remaining_dnsmasq"
        warning "可尝试手动清理: sudo killall hostapd dnsmasq"
    fi
     
    echo ""
    if [ "$stopped_any" = true ]; then
        success "✓ AP 已完全停止 (退出码: $EXIT_AP_STOPPED)"
        log_info "AP 已停止"
        echo ""
        info "日志文件已保留:"
        [ -f "$LOG_FILE" ] && echo "  - $LOG_FILE"
        [ -f "$HOSTAPD_LOG" ] && echo "  - $HOSTAPD_LOG"
        [ -f "$DNSMASQ_LOG" ] && echo "  - $DNSMASQ_LOG"
        [ -f "$ERROR_LOG" ] && echo "  - $ERROR_LOG"
        echo ""
        return "$EXIT_AP_STOPPED"
    else
        info "AP 未运行，无需停止 (退出码: $EXIT_AP_NOT_RUNNING)"
        log_info "AP 未运行，无需停止"
        echo ""
        info "日志文件位置:"
        [ -f "$LOG_FILE" ] && echo "  - $LOG_FILE"
        [ -f "$HOSTAPD_LOG" ] && echo "  - $HOSTAPD_LOG"
        [ -f "$DNSMASQ_LOG" ] && echo "  - $DNSMASQ_LOG"
        [ -f "$ERROR_LOG" ] && echo "  - $ERROR_LOG"
        echo ""
        return "$EXIT_AP_NOT_RUNNING"
    fi
}

# 清理函数（供信号处理器使用）
cleanup_on_interrupt() {
    echo ""
    warning "接收到中断信号，正在清理..."
    
    # 停止所有相关进程
    local hostapd_pids=$(ps aux | grep "[h]ostapd" | awk '{print $2}')
    local dnsmasq_pids=$(ps aux | grep "[d]nsmasq" | awk '{print $2}')
    
    [ -n "$hostapd_pids" ] && kill -KILL $hostapd_pids 2>/dev/null || true
    [ -n "$dnsmasq_pids" ] && kill -KILL $dnsmasq_pids 2>/dev/null || true
    
    # 清理 PID 文件
    rm -f "$PID_FILE" "$DNSMASQ_PID" 2>/dev/null || true
    
    # 清理锁文件
    rm -rf "$LOCK_FILE" 2>/dev/null || true
    
    warning "清理完成，AP 启动已中断"
    exit 130  # 130 = 128 + 2 (SIGINT)
}

# 启动 AP
start_ap() {
    # 设置信号处理器，使启动过程可以被中断
    trap cleanup_on_interrupt INT TERM
    
    info "初始化环境..."
    init_dirs
    
    # 停止现有 AP
    info "停止现有 AP（如果正在运行）..."
    check_and_kill_existing_processes

    # 停止系统 DHCP 服务
    stop_system_dhcp_services

    # 重新加载驱动（先卸载后加载）
    # reload_driver "$DRIVER_NAME" "reload"

    # 配置网络接口
    configure_interface "$INTERFACE" "$IP"
    
    # 生成 dnsmasq 配置
    generate_dnsmasq_config
    
    # 启动 dnsmasq
    info "启动 DHCP/DNS 服务 (dnsmasq)..."
    if dnsmasq -C "$DNSMASQ_CONF" -x "$DNSMASQ_PID" 2>&1 | tee -a "$LOG_FILE"; then
        if [ -f "$DNSMASQ_PID" ]; then
            success "✓ dnsmasq 启动成功 (PID: $(cat $DNSMASQ_PID))"
            log_info "dnsmasq 启动成功 (PID: $(cat $DNSMASQ_PID))"
        else
            warning "dnsmasq 启动但 PID 文件未创建"
        fi
    else
        log_error "dnsmasq 启动失败"
        error_exit "$EXIT_DNSMASQ_FAILED" "dnsmasq 启动失败，请查看日志: $DNSMASQ_LOG"
    fi
    
    # 启动 hostapd
    info "启动 hostapd..."
    if [ -n "$HOSTAPD_DEBUG_LEVEL" ]; then
        # 调试模式：不使用 -B，通过 & 在后台运行，将调试输出写入日志文件
        info "hostapd 调试级别: -$HOSTAPD_DEBUG_LEVEL (日志: $HOSTAPD_LOG)"
        nohup hostapd -"$HOSTAPD_DEBUG_LEVEL" "$HOSTAPD_CONF" >> "$HOSTAPD_LOG" 2>&1 &
        local bg_pid=$!
        echo "$bg_pid" > "$PID_FILE"
        sleep 1

        if kill -0 "$bg_pid" 2>/dev/null; then
            success "✓ hostapd 启动成功 (PID: $bg_pid, 调试级别: -$HOSTAPD_DEBUG_LEVEL)"
            log_info "hostapd 启动成功 (PID: $bg_pid, 调试级别: -$HOSTAPD_DEBUG_LEVEL)"
        else
            rm -f "$PID_FILE"
            log_error "hostapd 启动失败"
            echo "=== hostapd 错误日志 ===" >> "$ERROR_LOG"
            tail -20 "$HOSTAPD_LOG" >> "$ERROR_LOG"
            echo "======================" >> "$ERROR_LOG"

            stop_process "$DNSMASQ_PID" "dnsmasq"
            error_exit "$EXIT_HOSTAPD_FAILED" "hostapd 启动失败，详细信息请查看: $HOSTAPD_LOG"
        fi
    else
        # 普通模式：后台守护进程
        if hostapd -B -P "$PID_FILE" "$HOSTAPD_CONF" >> "$HOSTAPD_LOG" 2>&1; then
            local hostapd_pid=$(cat "$PID_FILE")
            success "✓ hostapd 启动成功 (PID: $hostapd_pid)"
            log_info "hostapd 启动成功 (PID: $hostapd_pid)"
        else
            log_error "hostapd 启动失败"
            echo "=== hostapd 错误日志 ===" >> "$ERROR_LOG"
            tail -20 "$HOSTAPD_LOG" >> "$ERROR_LOG"
            echo "======================" >> "$ERROR_LOG"

            stop_process "$DNSMASQ_PID" "dnsmasq"
            error_exit "$EXIT_HOSTAPD_FAILED" "hostapd 启动失败，详细信息请查看: $HOSTAPD_LOG"
        fi
    fi
    
    # 等待服务稳定
    sleep 2
    
    # 验证运行状态
    if [ -f "$PID_FILE" ] && kill -0 "$(cat $PID_FILE)" 2>/dev/null; then
        return 0
    else
        stop_process "$DNSMASQ_PID" "dnsmasq"
        log_error "AP 启动失败，hostapd 进程未能保持运行"
        error_exit "$EXIT_HOSTAPD_FAILED" "AP 启动失败，hostapd 进程未能保持运行"
    fi
}

# 主函数
main() {
    check_root
    check_dependencies
    
    # 解析参数（这会处理 --stop 选项并直接执行）
    parse_arguments "$@"
    
    # 如果执行到这里，说明不是 stop 模式
    # 获取锁，防止多个实例同时运行
    acquire_lock
    
    # 检查网卡支持
    info "检查网卡 $INTERFACE 对 $BAND GHz 的支持..."
    check_band_support "$INTERFACE" "$BAND"
    success "✓ 网卡支持检查通过"
    
    # 显示配置
    show_summary
    
    # 生成配置文件
    info "生成 hostapd 配置文件..."
    generate_hostapd_config
    success "✓ 配置文件已生成"
    
    echo ""
    info "启动 AP..."
    echo ""
    
    # 启动 AP
    if start_ap; then
        echo ""
        success "=================================================="
        success "✓ AP 启动成功！"
        success "=================================================="
        echo ""
        
        # 信道显示
        local channel_display="$CHANNEL"
        if [ "$CHANNEL" -eq 0 ]; then
            channel_display="自动选择 (ACS)"
        fi
        
        echo "  WiFi 名称: $SSID"
        echo "  WiFi 密码: $PASSWORD"
        echo "  频段:      $BAND GHz"
        echo "  信道:      $channel_display"
        echo "  网关:      $IP"
        echo ""
        echo "日志文件:"
        echo "  主日志:    $LOG_FILE"
        echo "  hostapd:   $HOSTAPD_LOG"
        echo "  dnsmasq:   $DNSMASQ_LOG"
        echo ""
        info "设备现在可以连接到此热点并自动获取 IP 地址"
        info "提示: 可使用 'sudo $0 -r' 快速重新启动相同配置"
        echo ""
        exit "$EXIT_SUCCESS"
    else
        error_exit "$EXIT_AP_START_FAILED" "AP 启动失败，请查看日志"
    fi
}

main "$@"
